cmake_minimum_required(VERSION 3.16)
project(qbudget_api LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(qbudget-api src/main.cpp)
target_compile_features(qbudget-api PRIVATE cxx_std_17)
target_include_directories(qbudget-api PRIVATE vendor)
target_link_libraries(qbudget-api PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(qbudget-api PRIVATE ws2_32 winhttp)
else()
    find_package(CURL REQUIRED)
    target_link_libraries(qbudget-api PRIVATE CURL::libcurl)
endif()
