# C++17 API and deterministic regression tests; platform-specific HTTPS implementation.
cmake_minimum_required(VERSION 3.16)
project(qbudget_api LANGUAGES CXX)
find_package(Threads REQUIRED)
include(CTest)
add_executable(qbudget-api src/main.cpp)
set(QBUDGET_TARGETS qbudget-api)
if(BUILD_TESTING)
    add_executable(qbudget-tests tests/data_test.cpp)
    list(APPEND QBUDGET_TARGETS qbudget-tests)
    add_test(NAME qbudget-domain COMMAND qbudget-tests)
endif()
if(NOT WIN32)
    find_package(CURL REQUIRED)
endif()
foreach(target IN LISTS QBUDGET_TARGETS)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE vendor)
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(${target} PRIVATE ws2_32 winhttp)
    else()
        target_link_libraries(${target} PRIVATE CURL::libcurl)
    endif()
endforeach()
